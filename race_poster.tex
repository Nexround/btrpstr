\begin{headerbox}{rda}{col=method1,header={01\quad Collect Evidence from Residual Stream},header font=\postersectionfont,
 header style={fill=pgreen,text=white},body style={fill=white},sep=.65cm,font=\fontsize{29}{36}\selectfont\raggedright}
Existing attribution approaches face a severe trade-off between computational tractability and functional validity. On one hand, gradient-based methods require iterative backward passes that incur prohibitive overhead, limiting scalable domain-wide auditing. On the other hand, raw activations fail to measure functional consistency due to three key flaws:
(1) Lack of direction;
(2) Orthogonal cancellation; and
(3) Masked instability.

\begin{headerbox}{cam}{col=method2,header={02\quad Estimate, Rank, and Filter},header font=\postersectionfont,

\textbf{Goal: Evaluating Functional Consistency}

A domain-specific neuron must satisfy two desiderata across the observation set $T_c$:
\begin{itemize}
\setlength{\topsep}{0em}
\setlength{\itemsep}{0em}
\setlength{\parsep}{0pt}
\setlength{\parskip}{0pt}
\item \textbf{Magnitude:} $\mathbb{E}_{t\in T_c}[e_j^{(t)}]$ is large and positive relative to peers in the module.
\item \textbf{Stability:} $\operatorname{Var}_{t\in T_c}[e_j^{(t)}]$ is small---not driven by rare outliers or sign fluctuations.
\end{itemize}

\subhead{Estimate | Bayesian Evidence Aggregation}
\textbf{Motivation:} Simple empirical averages discard cross-input variation and obscure sample-to-sample stability patterns.

\textbf{NIG Working Model:} Treat signed RDA evidence as noisy draws $e_j^{(t)}\sim\mathcal{N}(\mu_j,\sigma_j^2)$ with a conjugate Normal-Inverse-Gamma prior $\operatorname{NIG}(\mu_0,\lambda_0,\alpha_0,\beta_0)$.

\textbf{O(1) Streaming Inference:} Maintaining empirical mean $\bar e_j$ and centered sum of squares $SS_j$ yields a closed-form marginal Student's $t$-posterior for the mean evidence:
\[
\mu_j\mid\{e_j^{(t)}\}\sim t_{2\alpha_n}(\mu_{n,j},\,\sigma_{\mu,j}),\qquad \text{where}\quad
\sigma_{\mu,j}^2=\frac{\beta_{n,j}}{\alpha_n\lambda_n}.
\]
The scale $\sigma_{\mu,j}$ captures epistemic uncertainty---inflated by scarce, noisy, or sign-fluctuating evidence.

\subhead{Rank | Conservative Alignment Magnitude (CAM)}
\textbf{Motivation:} Weak or sparse activations can artificially inflate empirical SNR, while global sorting unfairly favors late layers where updates align closer to logits.

\textbf{Credible Lower Bound:} CAM scores neurons by a one-sided conservative $(1-\gamma)$-lower credible bound on positive mean evidence:
\[
\mathrm{CAM}_j(\gamma)=\mathbf{1}[\mu_{n,j}>0]\cdot\max\!\left(0,\,\mu_{n,j}-t_{1-\gamma,\,2\alpha_n}\cdot\sigma_{\mu,j}\right).
\]
\textbf{Stratified Budget:} Neurons are ranked by CAM within each layer and module to preserve coverage across the model's hierarchical computation.

\subhead{Filter | Reference-Set Filtering (RSF)}
\textbf{Motivation:} Under feature superposition and polysemanticity, raw target rankings conflate domain-specific functions with broadly shared general language capabilities.

\textbf{Dual-Corpus Filtering:} Traverses target ranking $\Pi_{\mathrm{tar}}^\ell$ and discards neurons that also rank highly on a general reference corpus (WikiText-2, $B_{\mathrm{ref}}^\ell$) to retain $K_{\mathrm{sel}}$ domain neurons:
\[
\Pi_{\mathrm{RSF}}^\ell=\left[j\in\Pi_{\mathrm{tar}}^\ell:j\notin B_{\mathrm{ref}}^\ell\right],\qquad
S_{\mathrm{spec}}^\ell=\operatorname{First}_{K_{\mathrm{sel}}}(\Pi_{\mathrm{RSF}}^\ell).
\]
Prevents baseline linguistic collapse during subsequent behavioral interventions.
